  1980	function drawSimCanvas(beta,omega0,theta0){
  1981	  const canvas=document.getElementById('sim_canvas');
  1982	  if(!canvas)return;
  1983	  const ctx=canvas.getContext('2d');
  1984	  const w=canvas.width,h=canvas.height;
  1985	  ctx.clearRect(0,0,w,h);
  1986	  const style=getComputedStyle(document.documentElement);
  1987	  const bg=style.getPropertyValue('--bg2').trim()||'#e9edf3';
  1988	  const col1=style.getPropertyValue('--a1').trim()||'#4f46e5';
  1989	  const col2=style.getPropertyValue('--a2').trim()||'#7c3aed';
  1990	  const border=style.getPropertyValue('--border').trim()||'#d4dae3';
  1991	  const t3=style.getPropertyValue('--t3').trim()||'#7d8da1';
  1992	  // Background
  1993	  ctx.fillStyle=bg;ctx.fillRect(0,0,w,h);
  1994	  // Grid
  1995	  ctx.strokeStyle=border;ctx.lineWidth=0.5;ctx.globalAlpha=0.3;
  1996	  for(let x=0;x<w;x+=w/10){ctx.beginPath();ctx.moveTo(x,0);ctx.lineTo(x,h);ctx.stroke();}
  1997	  for(let y=0;y<h;y+=h/6){ctx.beginPath();ctx.moveTo(0,y);ctx.lineTo(w,y);ctx.stroke();}
  1998	  // Zero line
  1999	  ctx.strokeStyle=t3;ctx.lineWidth=1;ctx.setLineDash([6,4]);ctx.globalAlpha=0.5;
  2000	  ctx.beginPath();ctx.moveTo(0,h/2);ctx.lineTo(w,h/2);ctx.stroke();ctx.setLineDash([]);
  2001	  // Compute & draw waveform
  2002	  const zeta=beta/omega0;
  2003	  const tMax=zeta<1? Math.min(20, 8/Math.max(beta,0.01)): Math.min(15, 5/Math.max(beta,0.01));
  2004	  const theta0Rad=theta0*Math.PI/180;
  2005	  // First pass: compute values and find max amplitude
  2006	  const vals=[];
  2007	  let maxAmp=0;
  2008	  for(let px=0;px<w;px++){
  2009	    const t=px/w*tMax;
  2010	    let theta;
  2011	    if(zeta<0.999){
  2012	      const wd=omega0*Math.sqrt(1-zeta*zeta);
  2013	      theta=theta0Rad*Math.exp(-beta*t)*Math.cos(wd*t);
  2014	    }else if(zeta<1.05){
  2015	      theta=theta0Rad*(1+omega0*t)*Math.exp(-omega0*t);
  2016	    }else{
  2017	      const sq=Math.sqrt(zeta*zeta-1);
  2018	      const l1=omega0*(zeta-sq),l2=omega0*(zeta+sq);
  2019	      const c1=theta0Rad*l2/(l2-l1),c2=-theta0Rad*l1/(l2-l1);
  2020	      theta=c1*Math.exp(-l1*t)+c2*Math.exp(-l2*t);
  2021	    }
  2022	    vals.push(theta);
  2023	    if(Math.abs(theta)>maxAmp)maxAmp=Math.abs(theta);
  2024	  }
  2025	  // Fixed scale: ±25°, fits 0-20° range nicely
  2026	  const ampScale=(h/2-30)/(25*Math.PI/180);
  2027	  ctx.beginPath();ctx.globalAlpha=0.9;
  2028	  const grad=ctx.createLinearGradient(0,0,w,0);
  2029	  grad.addColorStop(0,col1);grad.addColorStop(1,col2);
  2030	  ctx.strokeStyle=grad;ctx.lineWidth=2.5;
  2031	  for(let px=0;px<w;px++){
  2032	    const py=h/2-vals[px]*ampScale;
  2033	    px===0?ctx.moveTo(px,py):ctx.lineTo(px,py);
  2034	  }
  2035	  ctx.stroke();
  2036	  // Envelope
  2037	  if(zeta<1){
  2038	    ctx.beginPath();ctx.strokeStyle=col1;ctx.lineWidth=1;ctx.globalAlpha=0.3;ctx.setLineDash([4,4]);
  2039	    for(let px=0;px<w;px++){const t=px/w*tMax;const env=theta0Rad*Math.exp(-beta*t);const py=h/2-env*ampScale;px===0?ctx.moveTo(px,py):ctx.lineTo(px,py);}
  2040	    ctx.stroke();
  2041	    ctx.beginPath();
  2042	    for(let px=0;px<w;px++){const t=px/w*tMax;const env=-theta0Rad*Math.exp(-beta*t);const py=h/2-env*ampScale;px===0?ctx.moveTo(px,py):ctx.lineTo(px,py);}
  2043	    ctx.stroke();ctx.setLineDash([]);
  2044	  }
  2045	  // Labels
  2046	  ctx.globalAlpha=0.7;ctx.fillStyle=t3;ctx.font='22px Consolas,monospace';
  2047	  ctx.fillText('θ(t)',12,28);ctx.fillText('t (s)',w-60,h-10);
  2048	  ctx.fillStyle=col1;ctx.globalAlpha=0.5;ctx.font='20px "Segoe UI",sans-serif';
  2049	  ctx.fillText('θ₀='+theta0+'°  β='+beta.toFixed(2)+'  ω₀='+omega0.toFixed(1),12,h-12);
  2050	}
